.7, 2.45, 3.2, 4.0, 4.7, 5.5, 6.25];
  openModal(`<h3>Nhịp cồng chiêng</h3><p class="muted">Gõ khi thanh vàng chạm ô sáng. Cần 5 nhịp đúng.</p><div class="track" id="track"><div class="zone"></div></div><button type="button" id="hit">GÕ CHIÊNG</button><p id="rh-meta" class="muted"></p>`);
  const track = $("modal").querySelector("#track");
  times.forEach((_, i) => { const n = document.createElement("div"); n.className = "beat"; n.dataset.i = String(i); track.appendChild(n); });
  const t0 = performance.now();
  const lead = 1.6;
  const hit = new Set();
  let done = false;
  const judge = () => {
    const now = (performance.now() - t0) / 1000;
    let best = -1, bd = 0.2;
    times.forEach((t, i) => { if (hit.has(i)) return; const d = Math.abs(now - t); if (d < bd) { bd = d; best = i; } });
    if (best >= 0) { hit.add(best); playGong(); } else playWrong();
    const meta = $("modal").querySelector("#rh-meta");
    if (meta) meta.textContent = `${hit.size}/8 đúng`;
  };
  $("modal").querySelector("#hit").onclick = judge;
  state.mini = { tick() {
    if (done) return;
    const now = (performance.now() - t0) / 1000;
    track.querySelectorAll(".beat").forEach((n) => {
      const i = Number(n.dataset.i);
      const y = (now - (times[i] - lead)) / lead;
      n.style.top = Math.max(-10, Math.min(100, y * 100)) + "%";
      n.style.opacity = hit.has(i) ? "0.25" : "1";
    });
    if (now > times[times.length - 1] + 0.7) {
      done = true;
      if (hit.size >= 5) completeQuest("tn");
      else failMini("tn", "Chưa đủ nhịp. Cần ít nhất 5 tiếng đúng.");
    }
  } };
}

function tryPick(cx, cy) {
  if (state.mode !== "play" || state.paused || state.photo) return;
  const rect = renderer.domElement.getBoundingClientRect();
  ndc.x = ((cx - rect.left) / rect.width) * 2 - 1;
  ndc.y = -((cy - rect.top) / rect.height) * 2 + 1;
  raycaster.setFromCamera(ndc, camera);
  const hits = raycaster.intersectObjects([crowdBody, lanternBody], false);
  if (!hits.length) return;
  const h = hits[0];
  if (h.object === crowdBody && h.instanceId != null && npc[h.instanceId]?.x != null) openProfile(h.instanceId);
  else if (h.object === lanternBody && h.instanceId != null) {
    const w = state.wishes[h.instanceId % state.wishes.length];
    if (w) openModal(`<h3>Đèn của ${esc(w.name)}</h3><p>${esc(w.text)}</p>`);
  }
}

function shoot() {
  ensureAudio(); playShutter();
  renderer.render(scene, camera);
  const src = renderer.domElement;
  const c = document.createElement("canvas");
  const w = 480; const h = Math.round(480 * src.height / src.width);
  c.width = w; c.height = h;
  c.getContext("2d").drawImage(src, 0, 0, w, h);
  let url = "";
  try { url = c.toDataURL("image/jpeg", 0.62); } catch { toast("Không lưu được ảnh"); return; }
  state.photos.push({ url, place: state.place, at: Date.now() });
  if (state.photos.length > state.photoCap) state.photos.splice(0, state.photos.length - state.photoCap);
  const near = state.place !== "hub" && regions[state.place] && Math.hypot(playerPos.x - (regions[state.place].ox + 0.2), playerPos.z - (regions[state.place].oz - 7.4)) < 4;
  const bonus = near && !state.photoBonus[state.place];
  if (bonus) state.photoBonus[state.place] = true;
  gain(bonus ? 6 : 3, bonus ? 10 : 5, true);
  toast(bonus ? "Ảnh đẹp tại điểm săn mây! +6 hạt sen" : "Đã lưu vào bộ sưu tập");
  save();
  document.body.style.background = "#fff";
  setTimeout(() => { document.body.style.background = ""; }, 80);
}

function save() {
  const data = {
    v: 1, name: state.name, gender: state.gender, outfit: state.outfit, sen: state.sen, xp: state.xp,
    km: state.km, stamps: state.stamps, quests: state.quests, photos: state.photos.slice(-state.photoCap),
    wishes: state.wishes.slice(0, 40), friends: state.friends, gifts: state.gifts, vip: state.vip,
    music: state.music, sfx: state.sfx, photoCap: state.photoCap, photoBonus: state.photoBonus,
    quizDone: state.quizDone, wishDone: state.wishDone, items: state.items, certSeen: state.certSeen,
  };
  try { localStorage.setItem(KEY, JSON.stringify(data)); }
  catch {
    data.photos = data.photos.slice(-2);
    try { localStorage.setItem(KEY, JSON.stringify(data)); } catch { /* quota */ }
  }
}
function load() {
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return false;
    const d = JSON.parse(raw);
    Object.assign(state, {
      name: d.name || state.name, gender: d.gender || "nam", outfit: d.outfit || "bui",
      sen: d.sen || 0, xp: d.xp || 0, km: d.km || 0, stamps: d.stamps || {}, quests: d.quests || {},
      photos: d.photos || [], wishes: d.wishes?.length ? d.wishes : state.wishes, friends: d.friends || [],
      gifts: d.gifts || 0, vip: d.vip || {}, music: d.music ?? 0.35, sfx: d.sfx ?? 0.75,
      photoCap: d.photoCap || 6, photoBonus: d.photoBonus || {}, quizDone: d.quizDone || {},
      wishDone: d.wishDone || {}, items: d.items || {}, certSeen: !!d.certSeen,
    });
    return true;
  } catch { return false; }
}
