function buildHub() {
  hub = new THREE.Group();
  scene.add(hub);
  waterGeo = new THREE.PlaneGeometry(230, 230, 28, 28);
  waterGeo.rotateX(-Math.PI / 2);
  waterBaseY = new Float32Array(waterGeo.attributes.position.count);
  const arr = waterGeo.attributes.position.array;
  for (let i = 0; i < waterBaseY.length; i++) waterBaseY[i] = arr[i * 3 + 1];
  const water = new THREE.Mesh(waterGeo, matColor(0x3aa8bc, { opacity: 0.94 }));
  water.position.y = -0.45;
  hub.add(water);
  for (let i = 0; i < 9; i++) {
    const a = (i / 9) * Math.PI * 2;
    const h = 16 + (i % 3) * 7;
    addMesh(hub, cone(9 + (i % 4) * 2, h, 5), i % 2 ? 0x6ea35a : 0x87b86a, Math.cos(a) * 96, h / 2 - 4, Math.sin(a) * 96);
  }
  for (let i = 0; i < 5; i++) {
    const cloud = addMesh(hub, sph(4, 8, 6), 0xfffaf3, -40 + i * 22, 22 + (i % 2) * 3, -20 + (i % 3) * 8);
    cloud.scale.set(1.6, 0.45, 1);
    bobbers.push(cloud);
  }
  for (const isl of ISLANDS) {
    addMesh(hub, cyl(isl.r + 1.3, isl.r + 1.5, 0.5, 24), 0xe7c98a, isl.x, -0.15, isl.z);
    addMesh(hub, cyl(isl.r, isl.r, 0.7, 24), isl.color, isl.x, 0.15, isl.z);
    arch(hub, isl.x, isl.z, isl.id === "mt" ? 0xc5312a : 0xb94a3e);
    const lab = makeLabel(isl.name, 3.4);
    lab.position.set(isl.x, 3.5, isl.z);
    hub.add(lab);
  }
  for (let i = 0; i < ISLANDS.length - 1; i++) {
    const a = ISLANDS[i], b = ISLANDS[i + 1];
    const dx = b.x - a.x, dz = b.z - a.z;
    const len = Math.hypot(dx, dz);
    const m = addMesh(hub, box(4.2, 0.22, len), 0xd8c098, (a.x + b.x) / 2, 0.16, (a.z + b.z) / 2);
    m.rotation.y = Math.atan2(dx, dz);
  }
  // landmarks on the sa bàn
  for (let i = 0; i < 4; i++) addMesh(hub, box(3.2 - i * 0.45, 0.35, 1.4), i % 2 ? 0xe1b84a : 0x7fbf4a, -3, 0.55 + i * 0.32, -56 - i);
  stilt(hub, 6, -44);
  addMesh(hub, cone(3.2, 7, 5), 0x8d9a62, -8, 3.6, -58);
  temple(hub, 20, -20);
  jar(hub, 10, -12, 1.1); jar(hub, 11.2, -13, 0.8);
  tree(hub, 18, -10);
  house(hub, 6, 16, 1.6, 1.3, 1.1, 0xf2d48a, 0xb94a3e);
  addMesh(hub, sph(0.18, 8, 6), 0xffb347, 6, 2.5, 17.2, { emissive: 0xff5a00 });
  rong(hub, -16, 42);
  palm(hub, 14, 78, 1); boat(hub, 2, 82, 0x8a4b2a); umbrella(hub, 12, 70, 0xf25c4c);
  const flowerGeo = cone(0.12, 0.28, 5); flowerGeo.translate(0, 0.2, 0);
  const flowers = new THREE.InstancedMesh(flowerGeo, matColor(0xff7aa2), 36);
  flowers.frustumCulled = false;
  for (let i = 0; i < 36; i++) {
    const isl = ISLANDS[i % 5];
    const ang = i * 0.9;
    dummy.position.set(isl.x + Math.cos(ang) * (isl.r * 0.55), 0.45, isl.z + Math.sin(ang) * (isl.r * 0.55));
    dummy.scale.setScalar(0.8 + (i % 3) * 0.2);
    dummy.rotation.set(0, ang, 0);
    dummy.updateMatrix();
    flowers.setMatrixAt(i, dummy.matrix);
    flowers.setColorAt(i, new THREE.Color(i % 2 ? 0xff7aa2 : 0xffe08a));
  }
  flowers.instanceMatrix.needsUpdate = true;
  if (flowers.instanceColor) flowers.instanceColor.needsUpdate = true;
  hub.add(flowers);
}
