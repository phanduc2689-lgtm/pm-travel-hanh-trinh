, z: (Math.random() - 0.5) * 10, s: 0.4 + Math.random() * 0.8, ph: Math.random() * 6 });
  const F = mk(50, 0xffe58a, 0.12); flies = F.pts; flies.userData = { arr: F.pos, n: 50 };
  const B = mk(40, 0xffffff, 0.1); bubbles = B.pts; bubbles.userData = { arr: B.pos, n: 40 };

function updateParticles(dt) {
  if (petals) {
    const meta = petals.userData.meta, arr = petals.userData.arr;
    const cx = state.mode === "select" ? 0 : playerPos.x;
    const cz = state.mode === "select" ? 0 : playerPos.z;
    for (let i = 0; i < meta.length; i++) {
      const p = meta[i];
      p.y -= dt * p.s;
      if (p.y < 0) p.y = 4 + Math.random() * 2;
      arr[i * 3] = cx + p.x + Math.sin(animT + p.ph) * 0.4;
      arr[i * 3 + 1] = (state.mode === "select" ? 0.8 : 0.4) + p.y;
      arr[i * 3 + 2] = cz + p.z;
    }
    petals.geometry.attributes.position.needsUpdate = true;
    petals.visible = state.mode === "select" || state.place === "tb";
  }
  if (flies?.visible) {
    const arr = flies.userData.arr;
    const cx = playerPos.x, cz = playerPos.z;
    for (let i = 0; i < 50; i++) {
      arr[i * 3] = cx + Math.sin(animT * 0.6 + i) * 8;
      arr[i * 3 + 1] = 1 + (i % 5) * 0.4 + Math.sin(animT * 2 + i) * 0.2;
      arr[i * 3 + 2] = cz + Math.cos(animT * 0.5 + i * 1.3) * 8;
    }
    flies.geometry.attributes.position.needsUpdate = true;
  }
  if (bubbles) {
    bubbles.visible = state.place === "mn" && state.mode === "play";
    if (bubbles.visible) {
      const arr = bubbles.userData.arr;
      for (let i = 0; i < 40; i++) {
        const y = ((animT * 0.4 + i * 0.2) % 2);
        arr[i * 3] = playerPos.x + ((i % 8) - 4) * 0.8;
        arr[i * 3 + 1] = y;
        arr[i * 3 + 2] = playerPos.z + (Math.floor(i / 8) - 2) * 0.8;
      }
      bubbles.geometry.attributes.position.needsUpdate = true;
    }
  }
  if (waterGeo && hub && hub.visible) {
    const arr = waterGeo.attributes.position.array;
    for (let i = 0; i < waterBaseY.length; i++) {
      const x = arr[i * 3], z = arr[i * 3 + 2];
      arr[i * 3 + 1] = waterBaseY[i] + Math.sin(x * 0.12 + animT) * 0.1 + Math.cos(z * 0.1 + animT * 0.8) * 0.08;
    }
    waterGeo.attributes.position.needsUpdate = true;
  }
  for (const b of bobbers) {
    const base = b.userData.baseY ?? b.position.y;
    if (b.userData.baseY == null) b.userData.baseY = b.position.y;
    b.position.y = base + Math.sin(animT * 1.3 + b.position.x) * 0.08;
  }
  flames.forEach((f, i) => { f.scale.y = 0.85 + Math.sin(animT * 8 + i) * 0.25; });
}

function clearGroup(g) {
  for (const ch of [...g.children]) {
    g.remove(ch);
    if (ch.geometry) ch.geometry.dispose();
  }
}

function held(code) {
  const src = forcedKeys || keys;
  return src.has(code);
}
function readMove() {
  let ix = 0, iz = 0;
  if (held("KeyW") || held("ArrowUp")) iz += 1;
  if (held("KeyS") || held("ArrowDown")) iz -= 1;
  if (held("KeyD") || held("ArrowRight")) ix += 1;
  if (held("KeyA") || held("ArrowLeft")) ix -= 1;
  if (joy) { ix += joy.x; iz += -joy.y; }
  const m = Math.hypot(ix, iz);
  if (m > 1) { ix /= m; iz /= m; }
  return { ix, iz };
}
