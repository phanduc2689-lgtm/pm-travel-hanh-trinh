function esc(s) {

}

function matColor(hex, opts) {
  const o = opts || {};
  const k = `${hex}|${o.opacity ?? ""}|${o.emissive ?? ""}|${o.basic ? 1 : 0}`;
  if (!MATS[k]) {
    if (o.basic) {
      MATS[k] = new THREE.MeshBasicMaterial({ color: hex, transparent: o.opacity != null && o.opacity < 1, opacity: o.opacity ?? 1, depthWrite: !(o.opacity < 1) });
    } else {
      const m = new THREE.MeshLambertMaterial({
        color: hex,
        emissive: new THREE.Color(o.emissive || 0x000000),
        emissiveIntensity: o.emissive ? 0.85 : 0,
        transparent: o.opacity != null && o.opacity < 1,
        opacity: o.opacity ?? 1,
      });
      if (o.opacity != null && o.opacity < 1) m.depthWrite = false;
      MATS[k] = m;
    }
  }
  return MATS[k];
}

}

function roundRect(g, x, y, w, h, r) {
  g.beginPath();
  g.moveTo(x + r, y);
  g.arcTo(x + w, y, x + w, y + h, r);
  g.arcTo(x + w, y + h, x, y + h, r);
  g.arcTo(x, y + h, x, y, r);
  g.arcTo(x, y, x + w, y, r);
  g.closePath();
}
