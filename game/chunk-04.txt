  const w = window.innerWidth, h = window.innerHeight;
  const cap = w < 800 ? 1.5 : 2;
  renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, cap));
  renderer.setSize(w, h, false);

}
function updateSky(dt) {
  state.day = (state.day + dt * 0.012) % 1;
  const k = 0.5 - 0.5 * Math.cos(state.day * Math.PI * 2);
  if (k < 0.5) sky.copy(SKY_NIGHT).lerp(SKY_DUSK, k / 0.5);
  else sky.copy(SKY_DUSK).lerp(SKY_DAY, (k - 0.5) / 0.5);
  if (state.place === "tb") sky.lerp(SKY_DUSK, 0.22);
  if (state.place === "mt") sky.lerp(new THREE.Color(0xff7d5a), 0.18);
  if (state.place === "tn") sky.lerp(new THREE.Color(0x2f4a38), 0.16);
  sun.intensity = 0.42 + k * 0.95;
  hemi.intensity = 0.42 + k * 0.45;
  const ang = state.day * Math.PI * 2;
  const px = state.mode === "select" ? 0 : playerPos.x;
  const pz = state.mode === "select" ? 0 : playerPos.z;
  sun.position.set(px + Math.cos(ang) * 26, 16 + k * 24, pz + 12);
  sun.target.position.set(px, 0, pz);
  fill.position.set(px - 8, 12, pz + 14);
  fill.target.position.set(px, 1, pz);
  const ball = scene.getObjectByName("sunball");
  if (ball) {
    ball.visible = state.mode === "play";
    ball.position.set(px + Math.cos(ang) * 60, 22 + k * 20, pz + Math.sin(ang) * 28);
  }
  if (state.mode === "select") {
    scene.background = null;
    scene.fog = null;
    renderer.setClearColor(0x000000, 0);
  } else {
    scene.background = sky;
    scene.fog = fog;
    fog.color.copy(sky);
    fog.near = state.place === "hub" ? 28 : 12;
    fog.far = state.place === "hub" ? 125 : state.place === "tb" ? 42 : 56;
    renderer.setClearColor(sky, 1);
  }
  if (flies) flies.visible = state.mode === "play" && k < 0.42 && (state.place === "hub" || state.place === "tn");
}

/* --- WEB AUDIO SYNTHESIZER ENGINE --- */
function ensureAudio() {
  try {
    const AC = window.AudioContext || window.webkitAudioContext;
    if (!AC) return;
    if (!ac) {
      ac = new AC();
      master = ac.createGain(); master.connect(ac.destination);
      musicGain = ac.createGain(); sfxGain = ac.createGain();
      musicGain.connect(master); sfxGain.connect(master);
      const len = ac.sampleRate * 2;
      noiseBuf = ac.createBuffer(1, len, ac.sampleRate);
      const d = noiseBuf.getChannelData(0);
      for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
    }
    if (ac.state === "suspended") ac.resume();
    applyVol();
  } catch { ac = null; }
}
function applyVol() {
  if (!musicGain) return;
  musicGain.gain.value = state.music;
  sfxGain.gain.value = state.sfx;
}
function envGain(t, dur, peak, dest) {
  const g = ac.createGain();
  g.connect(dest || sfxGain);
  g.gain.setValueAtTime(0.0001, t);
  g.gain.exponentialRampToValueAtTime(Math.max(0.0002, peak), t + 0.02);
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
  return g;
}
function tone(freq, dur, type, peak, slide) {
  if (!ac) return;
  const t = ac.currentTime;
  const o = ac.createOscillator();
  o.type = type || "sine";
  o.frequency.setValueAtTime(freq, t);
  if (slide) o.frequency.exponentialRampToValueAtTime(slide, t + dur);
  const g = envGain(t, dur, peak * state.sfx);
  o.connect(g); o.start(t); o.stop(t + dur + 0.03);
}
function noise(dur, freq, peak, kind) {
  if (!ac) return;
  const src = ac.createBufferSource();
  src.buffer = noiseBuf;
  const f = ac.createBiquadFilter();
  f.type = kind || "bandpass";
  f.frequency.value = freq;
  const g = envGain(ac.currentTime, dur, peak * state.sfx);
  src.connect(f); f.connect(g);
  src.start(); src.stop(ac.currentTime + dur + 0.03);
}

function stopAmbient() {
  for (const n of ambientNodes) { try { n.stop?.(); n.disconnect?.(); } catch { /* already stopped */ } }
  ambientNodes = [];
}
function setAmbient(kind) {
  stopAmbient();
  if (!ac || state.music <= 0.001 || state.mode !== "play") return;
  const freqs = { hub: 196, tb: 220, bb: 246, mt: 262, tn: 174, mn: 233 };
  const o = ac.createOscillator();
  o.type = "sine";
  o.frequency.value = freqs[kind] || 196;
  const g = ac.createGain(); g.gain.value = 0.045;
  const lp = ac.createBiquadFilter(); lp.type = "lowpass"; lp.frequency.value = 700;
  o.connect(lp); lp.connect(g); g.connect(musicGain); o.start();
  const src = ac.createBufferSource();
  src.buffer = noiseBuf; src.loop = true;
  const bp = ac.createBiquadFilter();
  bp.type = kind === "mn" || kind === "mt" ? "lowpass" : "highpass";
  bp.frequency.value = kind === "tb" ? 1600 : kind === "mn" ? 420 : 900;
  const ng = ac.createGain();
  ng.gain.value = kind === "tn" ? 0.012 : 0.028;
  src.connect(bp); bp.connect(ng); ng.connect(musicGain); src.start();
  ambientNodes.push(o, src);
}
